schema 0
array ::= "[" space ( value ("," space value)* )? space "]"
boolean ::= ("true" | "false")
char ::= [^"\\\x7F\x00-\x1F] | [\\] (["\\bfnrt] | "u" [0-9a-fA-F]{4})
count ::= ref-defs-Item
count-kv ::= "\"count\"" space ":" space count
count-rest ::= ( "," space type-kv )? type-rest
decimal-part ::= [0-9]{1,16}
integral-part ::= [0] | [1-9] [0-9]{0,15}
null ::= "null"
number ::= ("-"? integral-part) ("." decimal-part)? ([eE] [-+]? integral-part)?
object ::= "{" space ( string ":" space value ("," space string ":" space value)* )? space "}"
ref-defs-Item ::= value
root ::= "{" space value-kv ( "," space ( count-kv count-rest | type-kv type-rest | x-y-kv ) )? space "}"
space ::= | " " | "\n"{1,2} [ \t]{0,20}
string ::= "\"" char* "\""
type ::= "[" space type-item ("," space type-item)? space "]"
type-item ::= "{" space type-item-type-kv ( "," space ( type-item-id-kv ) )? space "}"
type-item-id ::= "\"" (")a:" "_"+ "$"{2,} | "_1") "\""
type-item-id-kv ::= "\"id\"" space ":" space type-item-id
type-item-type-kv ::= "\"type\"" space ":" space string
type-kv ::= "\"type\"" space ":" space type
type-rest ::= ( "," space x-y-kv )?
value ::= object | array | string | number | boolean | null
value- ::= "\"" (([0-9]) "0"{0,2}) "\""
value-kv ::= "\"value\"" space ":" space value-
x-y ::= ("-" [1-9] [0-9]{0,15} | [0-7])
x-y-kv ::= "\"x-y\"" space ":" space x-y
end grammar
